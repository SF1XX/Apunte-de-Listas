INSERCIÓN - LISTAS SIMPLES
1- lista vacia
	prim = nil
	
	prim:= q
	*q.prox:= nil; (o *p.prox:= nil);
	
2- inserción antes del primer elemento
	prim = p
	
	prim:= q
	*q.prox:= p

3- inserción en el medio
	*p.dato > q*dato
	
	*ant.prox:= q
	*q.prox:= p

4- inserción en la ultima posición
	p = nil
	
	*ant.prox:= q
	*q.prox:= nil (o *q.prox:= p)

-------------------------------------------

Proceso
	nuevo(q)
	leer(*q.dato)
	si prim = nil entonces
		prim:= q
		*p.prox:= nil
	sino
	p:= prim
	ant:= nil
	mientras (p <> nil) y (*p.dato < *q.dato) hacer
		ant:= p
		p:= *p.prox
	fm
	si p = prim entonces
		prim:= q
		*q.prox:= p
	sino
		*ant.prox:= q
		*q.prox:= p
	fs
finaccion
